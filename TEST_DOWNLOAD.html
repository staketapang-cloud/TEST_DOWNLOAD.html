/* ============================================================ GOOGLE SLIDES → PDF CONVERTER HIGH QUALITY PNG → PDF ============================================================ ARSITEKTUR: GITHUB PAGES ↓ GOOGLE APPS SCRIPT ↓ GOOGLE DRIVE ↓ GOOGLE SLIDES ↓ LARGE PNG THUMBNAIL ↓ PDF ↓ BROWSER FITUR: - Scan folder Google Drive - Recursive scan - Deteksi Google Slides - Skip file yang sudah pernah dikonversi - Force Convert - Google Export PDF untuk file normal - HIGH QUALITY fallback untuk file besar - Thumbnail SIZE = LARGE - PNG langsung - TIDAK ADA PNG → JPG - 1 Google Slides = 1 PDF final - Browser membuat 1 PDF final - Tidak membuat PDF per slide - Chunk hanya untuk transport data - PDF tetap satu ============================================================ */ /* ============================================================ KONFIGURASI ============================================================ */ const APP_NAME = 'SLIDE TO PDF CONVERTER'; const MIME_GOOGLE_SLIDES = 'application/vnd.google-apps.presentation'; const MIME_PDF = 'application/pdf'; const MIME_PNG = 'image/png'; const CACHE_SKIP_KEY = 'CONVERTED_SLIDES_HISTORY_V1'; const MAX_HISTORY = 5000; /* PENTING: LARGE = kualitas thumbnail tertinggi yang tersedia melalui Slides API. Jangan ubah menjadi SMALL. */ const FALLBACK_THUMBNAIL_SIZE = 'LARGE'; /* Jumlah slide yang dikirim dalam satu request. LARGE PNG jauh lebih besar daripada SMALL. 8 dipilih agar: - request tidak terlalu berat - mengurangi risiko Failed to fetch - tetap cukup cepat */ const LARGE_PNG_CHUNK_SIZE = 8; /* ============================================================ RESPONSE JSON ============================================================ */ function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } /* ============================================================ GET SHEET / BASIC ============================================================ */ function doGet(e) { try { return jsonResponse({ success: true, app: APP_NAME, message: 'Google Slides → PDF Converter API aktif.', timestamp: new Date().toISOString() }); } catch (err) { return jsonResponse({ success: false, error: err.message }); } } /* ============================================================ POST ============================================================ */ function doPost(e) { try { let data = {}; /* JSON body */ if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (jsonError) { data = e.parameter || {}; } } else { data = e.parameter || {}; } const action = String(data.action || '') .trim() .toLowerCase(); /* -------------------------------------------------------- SCAN -------------------------------------------------------- */ if (action === 'scan_folder') { return jsonResponse( scanFolderAPI(data) ); } /* -------------------------------------------------------- SLIDE INFO -------------------------------------------------------- */ if (action === 'slide_info') { return jsonResponse( slideInfoAPI(data) ); } /* -------------------------------------------------------- NORMAL CONVERSION -------------------------------------------------------- */ if (action === 'convert_one') { return jsonResponse( convertOneAPI(data) ); } /* -------------------------------------------------------- LARGE PNG CHUNK -------------------------------------------------------- */ if (action === 'convert_chunk') { return jsonResponse( convertChunkAPI(data) ); } /* -------------------------------------------------------- MARK CONVERTED -------------------------------------------------------- */ if (action === 'mark_converted') { return jsonResponse( markConvertedAPI(data) ); } /* -------------------------------------------------------- HISTORY -------------------------------------------------------- */ if (action === 'get_history') { return jsonResponse( getHistoryAPI() ); } if (action === 'clear_history') { return jsonResponse( clearHistoryAPI() ); } return jsonResponse({ success: false, error: 'Action tidak dikenali: ' + action }); } catch (err) { return jsonResponse({ success: false, error: err.message, stack: err.stack || '' }); } } /* ============================================================ SCAN FOLDER ============================================================ */ function scanFolderAPI(data) { try { const folderId = extractDriveId( data.folderUrl || data.folderId ); if (!folderId) { throw new Error( 'Folder Google Drive tidak valid.' ); } const recursive = String(data.recursive || 'false') .toLowerCase() === 'true'; const files = []; scanDriveFolder_( folderId, recursive, files ); return { success: true, files: files, total: files.length }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ RECURSIVE DRIVE SCAN ============================================================ */ function scanDriveFolder_( folderId, recursive, result ) { const folder = DriveApp.getFolderById(folderId); const files = folder.getFiles(); while (files.hasNext()) { const file = files.next(); const mime = file.getMimeType(); if ( mime === MIME_GOOGLE_SLIDES ) { result.push({ id: file.getId(), name: file.getName(), mimeType: mime, size: file.getSize(), url: file.getUrl(), parents: folder.getId() }); } } if (!recursive) { return; } const folders = folder.getFolders(); while (folders.hasNext()) { const child = folders.next(); scanDriveFolder_( child.getId(), true, result ); } } /* ============================================================ EXTRACT DRIVE ID ============================================================ */ function extractDriveId(value) { if (!value) { return ''; } const text = String(value).trim(); /* URL: /folders/ID */ let match = text.match( /\/folders\/([a-zA-Z0-9_-]+)/ ); if (match) { return match[1]; } /* Google Slides: /d/ID/ */ match = text.match( /\/d\/([a-zA-Z0-9_-]+)/ ); if (match) { return match[1]; } /* ID langsung */ if ( /^[a-zA-Z0-9_-]{10,}$/.test(text) ) { return text; } return ''; } /* ============================================================ SLIDE INFO ============================================================ */ function slideInfoAPI(data) { try { const presentationId = String( data.presentationId || data.fileId || '' ).trim(); if (!presentationId) { throw new Error( 'presentationId tidak ditemukan.' ); } const presentation = Slides.Presentations.get( presentationId ); const slides = presentation.slides || []; const pageIds = slides.map(function(slide) { return slide.objectId; }); return { success: true, presentationId: presentationId, name: presentation.title || '', totalSlides: pageIds.length, pageIds: pageIds }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ NORMAL GOOGLE EXPORT ============================================================ */ function convertOneAPI(data) { try { const presentationId = String( data.presentationId || data.fileId || '' ).trim(); if (!presentationId) { throw new Error( 'presentationId tidak ditemukan.' ); } /* Google Slides export PDF. */ const url = 'https://docs.google.com/presentation/d/' + presentationId + '/export/pdf'; const response = UrlFetchApp.fetch( url, { muteHttpExceptions: true, followRedirects: true } ); const code = response.getResponseCode(); if (code >= 200 && code < 300) { const blob = response.getBlob() .setName( sanitizeFileName_( data.fileName || 'presentation' ) + '.pdf' ); const bytes = blob.getBytes(); return { success: true, tooLarge: false, fileName: blob.getName(), mimeType: MIME_PDF, size: bytes.length, base64: Utilities.base64Encode( bytes ) }; } /* Google memberikan export limit. */ const body = response.getContentText(); if ( body.indexOf( 'exportSizeLimitExceeded' ) !== -1 || body.indexOf( 'file is too large' ) !== -1 || body.indexOf( 'too large' ) !== -1 ) { return { success: true, tooLarge: true, message: 'File terlalu besar untuk Google Export. Gunakan HIGH QUALITY PNG.' }; } return { success: false, error: 'Google Export gagal. HTTP ' + code }; } catch (err) { /* Jangan langsung gagal. File besar akan dialihkan ke HIGH QUALITY PNG. */ const msg = String(err.message || err); if ( msg.indexOf( 'too large' ) !== -1 || msg.indexOf( 'exportSizeLimitExceeded' ) !== -1 ) { return { success: true, tooLarge: true, message: 'File besar → HIGH QUALITY PNG.' }; } return { success: false, error: msg }; } } /* ============================================================ CONVERT CHUNK ============================================================ */ function convertChunkAPI(data) { try { const presentationId = String( data.presentationId || data.fileId || '' ).trim(); if (!presentationId) { throw new Error( 'presentationId tidak ditemukan.' ); } let pageIds = data.pageIds || []; /* Jika dikirim sebagai JSON string. */ if (typeof pageIds === 'string') { try { pageIds = JSON.parse(pageIds); } catch (err) { pageIds = []; } } if ( !Array.isArray(pageIds) || pageIds.length === 0 ) { throw new Error( 'pageIds kosong.' ); } /* HANYA mengambil thumbnail. Tidak membuat PDF di server. PDF final dibuat oleh browser. */ const images = getSlideThumbnailPNGBatch_( presentationId, pageIds ); return { success: true, presentationId: presentationId, count: images.length, thumbnailSize: FALLBACK_THUMBNAIL_SIZE, format: 'PNG', images: images }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ GET LARGE PNG THUMBNAILS ============================================================ */ function getSlideThumbnailPNGBatch_( presentationId, pageIds ) { const result = []; pageIds.forEach(function(pageId) { try { /* Slides API thumbnail. LARGE = kualitas tinggi. */ const thumbnail = Slides.Presentations.Pages .getThumbnail( presentationId, pageId, { 'thumbnailProperties.thumbnailSize': FALLBACK_THUMBNAIL_SIZE, 'thumbnailProperties.mimeType': MIME_PNG } ); const contentUrl = thumbnail.contentUrl; if (!contentUrl) { throw new Error( 'contentUrl thumbnail kosong untuk slide ' + pageId ); } /* Download PNG asli. */ const response = UrlFetchApp.fetch( contentUrl, { muteHttpExceptions: true } ); const code = response.getResponseCode(); if (code < 200 || code >= 300) { throw new Error( 'Gagal mengambil PNG slide ' + pageId + '. HTTP ' + code ); } const blob = response.getBlob(); const bytes = blob.getBytes(); /* Kirim ukuran gambar ke browser bila tersedia. Kita tidak mengubah gambar. */ result.push({ pageId: pageId, mimeType: MIME_PNG, size: bytes.length, base64: Utilities.base64Encode( bytes ) }); } catch (err) { result.push({ pageId: pageId, error: err.message }); } }); return result; } /* ============================================================ MARK CONVERTED ============================================================ */ function markConvertedAPI(data) { try { const id = String( data.presentationId || data.fileId || '' ).trim(); if (!id) { throw new Error( 'presentationId tidak ditemukan.' ); } const name = String( data.fileName || data.name || id ); const props = PropertiesService .getScriptProperties(); let history = props.getProperty( CACHE_SKIP_KEY ); let list = []; if (history) { try { list = JSON.parse(history); } catch (err) { list = []; } } /* Hapus ID lama jika ada. */ list = list.filter(function(item) { return item.id !== id; }); /* Masukkan paling baru ke bagian depan. */ list.unshift({ id: id, name: name, convertedAt: new Date().toISOString() }); /* Batasi history. */ if ( list.length > MAX_HISTORY ) { list = list.slice( 0, MAX_HISTORY ); } props.setProperty( CACHE_SKIP_KEY, JSON.stringify(list) ); return { success: true, id: id, message: 'File ditandai sudah dikonversi.' }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ HISTORY ============================================================ */ function getHistoryAPI() { try { const props = PropertiesService .getScriptProperties(); const history = props.getProperty( CACHE_SKIP_KEY ); let list = []; if (history) { try { list = JSON.parse(history); } catch (err) { list = []; } } return { success: true, history: list }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ CLEAR HISTORY ============================================================ */ function clearHistoryAPI() { try { PropertiesService .getScriptProperties() .deleteProperty( CACHE_SKIP_KEY ); return { success: true, message: 'History conversion berhasil dihapus.' }; } catch (err) { return { success: false, error: err.message }; } } /* ============================================================ SANITIZE FILE NAME ============================================================ */ function sanitizeFileName_(name) { return String(name || 'file') .replace(/[\\\/:*?"<>|#%{}]/g, '_') .replace(/\s+/g, ' ') .trim() .substring(0, 180); } /* ============================================================ TEST FUNCTION ============================================================ */ function TEST_PING() { Logger.log( JSON.stringify({ success: true, app: APP_NAME, engine: 'HIGH QUALITY LARGE PNG → PDF', thumbnail: FALLBACK_THUMBNAIL_SIZE, format: 'PNG', chunkSize: LARGE_PNG_CHUNK_SIZE, timestamp: new Date().toISOString() }) ); }
